## E2. content_landed's FIRST test is an ancestry test (spira/lib.sh).
##     A branch with no commits on it IS an ancestor of the base, so an aeon that
##     changed nothing gets the same answer as an aeon whose work merged.
content_landed() {
    local repo="$1" br="$2" base="$3" merged basetree
    git -C "$repo" merge-base --is-ancestor "$br" "$base" 2>/dev/null && return 0
    merged="$(git -C "$repo" merge-tree --write-tree "$base" "$br" 2>/dev/null)" || return 1
    merged="${merged%%$'\n'*}"
    [ -n "$merged" ] || return 1
    basetree="$(git -C "$repo" rev-parse "$base^{tree}" 2>/dev/null)" || return 1
    [ "$merged" = "$basetree" ]
}
